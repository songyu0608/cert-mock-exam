extract_metadata = {
    "name": "extract_metadata",
    "strict": True,
    "input_schema": METADATA_SCHEMA}

response = client.messages.create(
    model=MODEL, max_tokens=1024,
    system="Call extract_metadata for every attachment.",
    tools=[extract_metadata],

    messages=messages)
